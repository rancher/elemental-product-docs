:description: Create a registration form by using example random template variables in YAML format to define fields and structure data for user input
.registration example Random template variables
[,yaml]
----
apiVersion: elemental.cattle.io/v1beta1
kind: MachineRegistration
metadata:
  name: fire-nodes
  namespace: fleet-default
spec:
  machineName: "fire-node-${Random/Hex/12}"
  config:
    cloud-config:
      users:
        - name: root
          passwd: root
    elemental:
      install:
        reboot: true
        device: /dev/sda
        debug: true
  machineInventoryLabels:
    elemental.cattle.io/UUID: "${Random/UUID}"
