:description: Configure a node selector to control where Kubernetes pods run by editing the upgrade-node-selector.yaml file
.upgrade-node-selector.yaml
[,yaml]
----
apiVersion: elemental.cattle.io/v1beta1
kind: ManagedOSImage
metadata:
  name: my-upgrade
  namespace: fleet-default
spec:
  # Set to the new {elemental-product-name} version you would like to upgrade to
  osImage: "registry.suse.com/suse/sle-micro/5.5:latest"
  clusterTargets:
    - clusterName: my-cluster
  nodeSelector:
    matchLabels:
      kubernetes.io/hostname: my-machine
----
