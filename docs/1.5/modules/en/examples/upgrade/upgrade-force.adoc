:description: Configure the upgrade-force.yaml file to define how the system handles forced upgrades by specifying rules and conditions that determine when and how updates occur
.upgrade-force.yaml
[,yaml]
----
apiVersion: elemental.cattle.io/v1beta1
kind: ManagedOSImage
metadata:
  name: my-upgrade
  namespace: fleet-default
spec:
  # Set to the new {elemental-product-name} version you would like to upgrade to
  osImage: "registry.suse.com/suse/sle-micro/5.5:latest"
  clusterTargets:
    - clusterName: my-cluster
  upgradeContainer:
    envs:
      - name: FORCE
        value: "true"
----
